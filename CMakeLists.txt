cmake_minimum_required(VERSION 3.20)
project(fx3_cyapi LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(PYBIND11_FINDPYTHON ON)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 CONFIG REQUIRED)
find_package(Threads REQUIRED)
pybind11_add_module(_native native/engine.cpp native/bindings.cpp)
target_include_directories(_native PRIVATE native)
target_link_libraries(_native PRIVATE Threads::Threads)
if(WIN32)
  if(NOT MSVC)
    message(FATAL_ERROR "Windows builds require MSVC")
  endif()
  set(CYAPI_DIR "$ENV{CYAPI_DIR}" CACHE PATH "External official CyAPI directory containing src/ and inc/")
  if(NOT CYAPI_DIR)
    # Optional local SDK; ignored by Git and excluded from source distributions.
    set(CYAPI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/CyUSB3_USB_Suite_Source/CyAPI_Library_Source/CyAPI/library/cpp")
  endif()
  if(NOT EXISTS "${CYAPI_DIR}/src/CyAPI.cpp" OR NOT EXISTS "${CYAPI_DIR}/inc/CyAPI.h")
    message(FATAL_ERROR "Official CyAPI sources are external. Set CYAPI_DIR to the directory containing src/CyAPI.cpp and inc/CyAPI.h, or pass -DCYAPI_DIR=<path>.")
  endif()
  add_library(cyapi_native STATIC "${CYAPI_DIR}/src/CyAPI.cpp")
  target_include_directories(cyapi_native PUBLIC "${CYAPI_DIR}/inc")
  target_compile_definitions(cyapi_native PRIVATE _CRT_SECURE_NO_WARNINGS _LIB)
  target_link_libraries(cyapi_native PUBLIC setupapi user32 uuid)
  target_sources(_native PRIVATE native/usb_windows.cpp)
  target_compile_definitions(_native PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(_native PRIVATE cyapi_native)
else()
  message(STATUS "Non-Windows build: simulation backend only")
endif()
install(TARGETS _native LIBRARY DESTINATION fx3_cyapi RUNTIME DESTINATION fx3_cyapi)
